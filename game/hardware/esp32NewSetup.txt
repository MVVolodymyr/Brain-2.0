// ESP32 Brain Ring Button Controller
// Updated setup without hardcoded secrets

#include <WiFi.h>
#include <PubSubClient.h>
#include <ArduinoJson.h>
#include <Preferences.h>

Preferences preferences;

// ==== WiFi Налаштування ====
const char* ssid = "YOUR_WIFI_SSID"; // ЗАМІНІТЬ НА ВАШЕ ІМ'Я МЕРЕЖІ
const char* password = "YOUR_WIFI_PASSWORD"; // ЗАМІНІТЬ НА ВАШ ПАРОЛЬ

// ==== MQTT Налаштування (Хардкодні, крім Client ID) ====
const int mqttPort = 1883; // Порт MQTT зазвичай фіксований для незашифрованого з'єднання
const char* MQTT_SERVER = "io.adafruit.com";
const char* MQTT_USERNAME = "vmishchenko"; // ВАШЕ ІМ'Я КОРИСТУВАЧА ADAFRUIT IO
const char* MQTT_AIO_KEY = "YOUR_ADAFRUIT_IO_KEY_HERE"; // ВАШ AIO KEY
// MQTT_CLIENT_ID тепер буде братися зі storedMqttPayload
const char* MQTT_TOPIC_BUTTON = "vmishchenko/feeds/esp32-button"; // Топік для публікації кнопки

WiFiClient espClient;
PubSubClient client(espClient);

// ==== Кнопка та LED ====
const int BUTTON_PIN = 0; // GPIO0 (BOOT кнопка)
const int LED_PIN = 2;    // Вбудований LED
const int TEAM_LED_PIN = 4; // Додатковий LED для команди

// ==== Змінні стану ====
bool lastButtonState = HIGH;
bool currentButtonState = HIGH;
unsigned long lastDebounceTime = 0;
unsigned long debounceDelay = 50;
String currentTeam = "red"; // Команда за замовчуванням
String storedMqttPayload = ""; // Зберігаємо MQTT payload для Client ID

void setup() {
    Serial.begin(115200);
    
    // Ініціалізація пінів
    pinMode(BUTTON_PIN, INPUT_PULLUP);
    pinMode(LED_PIN, OUTPUT);
    pinMode(TEAM_LED_PIN, OUTPUT);
    
    // Завантаження збережених даних
    preferences.begin("brainring", false);
    currentTeam = preferences.getString("team", "red");
    storedMqttPayload = preferences.getString("mqttPayload", "");
    preferences.end();
    
    // Підключення до WiFi
    connectToWiFi();
    
    // Налаштування MQTT
    client.setServer(MQTT_SERVER, mqttPort);
    client.setCallback(callback);
    
    Serial.println("ESP32 Brain Ring Controller Ready!");
    Serial.println("Current team: " + currentTeam);
}

void loop() {
    // Перевірка з'єднання WiFi
    if (WiFi.status() != WL_CONNECTED) {
        connectToWiFi();
    }
    
    // Перевірка з'єднання MQTT
    if (!client.connected()) {
        reconnect();
    }
    
    client.loop();
    
    // Обробка кнопки
    handleButton();
    
    delay(10);
}

void connectToWiFi() {
    Serial.print("Connecting to WiFi");
    WiFi.begin(ssid, password);
    
    while (WiFi.status() != WL_CONNECTED) {
        delay(500);
        Serial.print(".");
    }
    
    Serial.println();
    Serial.println("WiFi connected!");
    Serial.println("IP address: " + WiFi.localIP().toString());
}

void reconnect() {
    while (!client.connected()) {
        Serial.print("Attempting MQTT connection...");
        
        // Використовуємо збережений payload як Client ID, якщо він є
        String clientId = storedMqttPayload.length() > 0 ? storedMqttPayload : "ESP32Client";
        
        if (client.connect(clientId.c_str(), MQTT_USERNAME, MQTT_AIO_KEY)) {
            Serial.println("connected!");
            client.subscribe(MQTT_TOPIC_BUTTON);
        } else {
            Serial.print("failed, rc=");
            Serial.print(client.state());
            Serial.println(" try again in 5 seconds");
            delay(5000);
        }
    }
}

void callback(char* topic, byte* payload, unsigned int length) {
    String message = "";
    for (int i = 0; i < length; i++) {
        message += (char)payload[i];
    }
    
    Serial.println("Received MQTT message: " + message);
    
    // Парсинг JSON payload
    DynamicJsonDocument doc(1024);
    deserializeJson(doc, message);
    
    if (doc.containsKey("team")) {
        String newTeam = doc["team"];
        if (newTeam != currentTeam) {
            currentTeam = newTeam;
            
            // Збереження нової команди
            preferences.begin("brainring", false);
            preferences.putString("team", currentTeam);
            preferences.end();
            
            Serial.println("Team changed to: " + currentTeam);
            
            // Візуальна індикація зміни команди
            blinkTeamLED();
        }
    }
    
    // Збереження payload для використання як Client ID
    if (message != storedMqttPayload) {
        storedMqttPayload = message;
        preferences.begin("brainring", false);
        preferences.putString("mqttPayload", storedMqttPayload);
        preferences.end();
    }
}

void handleButton() {
    int reading = digitalRead(BUTTON_PIN);
    
    if (reading != lastButtonState) {
        lastDebounceTime = millis();
    }
    
    if ((millis() - lastDebounceTime) > debounceDelay) {
        if (reading != currentButtonState) {
            currentButtonState = reading;
            
            if (currentButtonState == LOW) { // Кнопка натиснута
                Serial.println("Button pressed for team: " + currentTeam);
                
                // Візуальна індикація
                digitalWrite(LED_PIN, HIGH);
                digitalWrite(TEAM_LED_PIN, HIGH);
                
                // Відправка MQTT повідомлення
                sendTeamSignal();
                
                delay(100); // Коротка затримка для візуального ефекту
                digitalWrite(LED_PIN, LOW);
                digitalWrite(TEAM_LED_PIN, LOW);
            }
        }
    }
    
    lastButtonState = reading;
}

void sendTeamSignal() {
    DynamicJsonDocument doc(1024);
    doc["team"] = currentTeam;
    doc["timestamp"] = millis();
    doc["action"] = "button_press";
    
    String payload;
    serializeJson(doc, payload);
    
    if (client.publish(MQTT_TOPIC_BUTTON, payload.c_str())) {
        Serial.println("MQTT message sent: " + payload);
    } else {
        Serial.println("Failed to send MQTT message");
    }
}

void blinkTeamLED() {
    for (int i = 0; i < 3; i++) {
        digitalWrite(TEAM_LED_PIN, HIGH);
        delay(200);
        digitalWrite(TEAM_LED_PIN, LOW);
        delay(200);
    }
}
